// src/components/ExpedientesCoordinacion.jsx

import React, { useState, useEffect } from "react";
import {
  Typography,
  Button,
  Paper,
  Alert,
  MenuItem,
  Select,
  InputLabel,
  FormControl,
  CircularProgress
} from "@mui/material";
import { DataGrid } from "@mui/x-data-grid";
import ExportExcel from "./ExportExcel";

function ExpedientesCoordinacion() {
  const [departamento, setDepartamento] = useState("");
  const [idCarga, setIdCarga] = useState("");
  const [departamentos, setDepartamentos] = useState([]);
  const [cargas, setCargas] = useState([]);
  const [rows, setRows] = useState([]);
  const [mensaje, setMensaje] = useState("");
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    fetch("http://localhost:3000/departamentos")
      .then(res => res.json())
      .then(data => setDepartamentos(data));

    fetch("http://localhost:3000/expedientes/cargas")
      .then(res => res.json())
      .then(data => setCargas(data));
  }, []);

  const handleBuscar = async () => {
    if (!departamento || !idCarga) {
      setMensaje("Debe seleccionar departamento e id_carga");
      return;
    }

    setLoading(true);
    try {
      const res = await fetch(
        `http://localhost:3000/expedientes/coordinacion/${idCarga}/${departamento}`
      );
      const data = await res.json();
      setRows(data);
      setMensaje(`Se encontraron ${data.length} registros`);
    } catch (err) {
      console.error(err);
      setMensaje("Error al buscar expedientes");
    } finally {
      setLoading(false);
    }
  };

  const columns = [
    { field: "rub", headerName: "RUB", width: 120 },
    { field: "departamento", headerName: "Departamento", width: 150 },
    { field: "municipio", headerName: "Municipio", width: 150 },
    { field: "estatus", headerName: "Estatus", width: 120 },
    { field: "nombre", headerName: "Nombre Completo", width: 250 },
  ];

  const exportColumns =
    rows.length > 0
      ? Object.keys(rows[0]).filter(
          key => key !== "id" && key !== "created_at" && key !== "updated_at"
        )
      : [];

  return (
    <Paper sx={{ p: 3, maxWidth: 900, margin: "auto", mt: 4 }}>
      <Typography variant="h5" sx={{ color: "var(--guatemala-dark)", mb: 2 }}>
        📑 Expedientes por Coordinación
      </Typography>

      <FormControl sx={{ minWidth: 200, mr: 2 }} size="small">
        <InputLabel>Departamento</InputLabel>
        <Select value={departamento} onChange={(e) => setDepartamento(e.target.value)}>
          {departamentos.map((d) => (
            <MenuItem key={d.id_departamento} value={d.nombre}>
              {d.nombre}
            </MenuItem>
          ))}
        </Select>
      </FormControl>

      <FormControl sx={{ minWidth: 150, mr: 2 }} size="small">
        <InputLabel>Id_Carga</InputLabel>
        <Select value={idCarga} onChange={(e) => setIdCarga(e.target.value)}>
          {cargas.map((c) => (
            <MenuItem key={c.id_carga} value={c.id_carga}>
              {c.id_carga}
            </MenuItem>
          ))}
        </Select>
      </FormControl>

      <Button className="editar" onClick={handleBuscar}>
        Buscar
      </Button>

      {rows.length > 0 && (
        <div style={{ marginTop: "15px" }}>
          <ExportExcel
            data={rows}
            columns={exportColumns.map((col) => ({
              header: col.toUpperCase(),
              key: col,
            }))}
            fileName="expedientes"
            className="excel"
          />
        </div>
      )}

      {mensaje && (
        <Alert severity={mensaje.includes("Error") ? "error" : "info"} sx={{ mt: 2 }}>
          {mensaje}
        </Alert>
      )}

      {loading ? (
        <div style={{ textAlign: "center", marginTop: "20px" }}>
          <CircularProgress />
          <Typography variant="body2" sx={{ mt: 1 }}>
            Cargando expedientes...
          </Typography>
        </div>
      ) : (
        <DataGrid
          rows={rows}
          columns={columns}
          autoHeight
          pageSize={10}
          rowsPerPageOptions={[10, 20, 50]}
          sx={{ mt: 2 }}
        />
      )}
    </Paper>
  );
}

export default ExpedientesCoordinacion;

// src/components/ActualizarExpedientes.jsx

import React, { useState, useEffect } from "react";
import {
  Typography,
  Button,
  Paper,
  Alert,
  Checkbox,
  Table,
  TableBody,
  TableCell,
  TableContainer,
  TableHead,
  TableRow,
  MenuItem,
  Select,
  InputLabel,
  FormControl
} from "@mui/material";

function ActualizarExpedientes() {
  const [departamento, setDepartamento] = useState("");
  const [municipio, setMunicipio] = useState("");
  const [departamentos, setDepartamentos] = useState([]);
  const [municipios, setMunicipios] = useState([]);
  const [rows, setRows] = useState([]);
  const [seleccionados, setSeleccionados] = useState([]);
  const [mensaje, setMensaje] = useState("");

  useEffect(() => {
    fetch("http://localhost:3000/departamentos")
      .then((res) => res.json())
      .then((data) => setDepartamentos(data || []));
  }, []);

  const handleDepartamentoChange = async (idDepartamento) => {
    setDepartamento(idDepartamento);
    setMunicipio("");
    const res = await fetch(`http://localhost:3000/municipios/${idDepartamento}`);
    const data = await res.json();
    setMunicipios(data || []);
  };

  const handleBuscar = async () => {
    if (!departamento || !municipio) {
      setMensaje("Debe seleccionar departamento y municipio");
      return;
    }

    try {
      const nombreDepto = departamentos.find(d => d.id_departamento === departamento)?.nombre;
      const res = await fetch(
        `http://localhost:3000/expedientes/filtrar?departamento=${nombreDepto.toUpperCase()}&municipio=${municipio.toUpperCase()}`
      );
      const data = await res.json();

      const filtrados = (data || [])
        .filter(r => r.estatus && ["Pendiente", "Nuevo"].includes(r.estatus.trim()))
        .map(r => ({
          ...r,
          nombre: [
            r.primer_nombre,
            r.segundo_nombre,
            r.tercer_nombre,
            r.primer_apellido,
            r.segundo_apellido,
            r.apellido_casada,
          ].filter(Boolean).join(" "),
        }));

      setRows(filtrados);
      setMensaje(`Se encontraron ${filtrados.length} registros`);
    } catch (err) {
      setMensaje("Error al buscar expedientes");
    }
  };

  const toggleSeleccion = (id) => {
    setSeleccionados((prev) =>
      prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id]
    );
  };

const handleActualizar = async () => {
  if (seleccionados.length === 0) {
    setMensaje("Debe seleccionar al menos un expediente");
    return;
  }

  try {
    const payload = {
      ids: seleccionados.map(id => Number(id)),
      nuevoEstatus: "Trabajado"
    };

    const res = await fetch("http://localhost:3000/expedientes_estatus/actualizar-estatus", {
      method: "PUT",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload)
    });

    const result = await res.json();

    if (res.ok && result.success) {
      const cantidad = result.cantidad || 0;
      setMensaje(`✅ Se actualizaron ${cantidad} registros correctamente`);

      // 👇 Reinicia el contador
     setSeleccionados([]);      
     
     // 👇 Espera 2 segundos antes de refrescar
     setTimeout(() => {
        handleBuscar();
     }, 5000);    } 
    else {
      setMensaje(result.error || result.mensaje || "Error al actualizar estatus");
    }
  } catch (err) {
    console.error("❌ Error de conexión:", err);
    setMensaje("Error de conexión al actualizar");
  }
};

  return (
    <Paper sx={{ p: 3, maxWidth: 950, margin: "auto", mt: 4 }}>
      <Typography variant="h5" sx={{ mb: 2 }}>
        🔄 Actualizar Expedientes
      </Typography>

      <FormControl sx={{ minWidth: 200, mr: 2 }} size="small">
        <InputLabel>Departamento</InputLabel>
        <Select
          value={departamento}
          onChange={(e) => handleDepartamentoChange(e.target.value)}
        >
          {departamentos.map((d) => (
            <MenuItem key={d.id_departamento} value={d.id_departamento}>
              {d.nombre}
            </MenuItem>
          ))}
        </Select>
      </FormControl>

      <FormControl sx={{ minWidth: 200, mr: 2 }} size="small">
        <InputLabel>Municipio</InputLabel>
        <Select
          value={municipio}
          onChange={(e) => setMunicipio(e.target.value)}
        >
          {municipios.map((m) => (
            <MenuItem key={m.id_municipio} value={m.nombre}>
              {m.nombre}
            </MenuItem>
          ))}
        </Select>
      </FormControl>

      <Button variant="contained" onClick={handleBuscar}>
        Buscar
      </Button>

      {rows.length > 0 && (
        <Button variant="contained" sx={{ ml: 2 }} onClick={handleActualizar}>
          Actualizar Estatus
        </Button>
      )}

      <Typography variant="subtitle1" sx={{ mt: 2, mb: 1, fontWeight: "bold" }}>
        📌 Registros seleccionados: {seleccionados.length}
      </Typography>

      {mensaje && (
        <Alert severity={mensaje.includes("Error") ? "error" : "success"} sx={{ mb: 2 }}>
          {mensaje}
        </Alert>
      )}

      <TableContainer>
        <Table>
          <TableHead>
            <TableRow>
              <TableCell>✔</TableCell>
              <TableCell>ID</TableCell>
              <TableCell>RUB</TableCell>
              <TableCell>Departamento</TableCell>
              <TableCell>Municipio</TableCell>
              <TableCell>Estatus</TableCell>
              <TableCell>Nombre Completo</TableCell>
            </TableRow>
          </TableHead>
          <TableBody>
            {rows.map((row) => (
              <TableRow key={row.id}>
                <TableCell>
                  <Checkbox
                    checked={seleccionados.includes(row.id)}
                    onChange={() => toggleSeleccion(row.id)}
                  />
                </TableCell>
                <TableCell>{row.id}</TableCell>
                <TableCell>{row.rub}</TableCell>
                <TableCell>{row.departamento}</TableCell>
                <TableCell>{row.municipio}</TableCell>
                <TableCell>{row.estatus}</TableCell>
                <TableCell>{row.nombre}</TableCell>
              </TableRow>
            ))}
          </TableBody>
        </Table>
      </TableContainer>
    </Paper>
  );
}

export default ActualizarExpedientes;

// src/components/Dashboard.jsx
import React, { useEffect, useState } from "react";
import { Bar, Pie } from "react-chartjs-2";
import {
  Chart as ChartJS,
  Title,
  Tooltip,
  Legend,
  BarElement,
  ArcElement,
  CategoryScale,
  LinearScale,
} from "chart.js";

ChartJS.register(
  Title,
  Tooltip,
  Legend,
  BarElement,
  ArcElement,
  CategoryScale,
  LinearScale
);

function Dashboard() {
  const [expedientesDepto, setExpedientesDepto] = useState([]);
  const [expedientesEstatus, setExpedientesEstatus] = useState([]);

  useEffect(() => {
    // carga inicial: todos los departamentos
    fetch("http://localhost:3000/expedientes/deptos")
      .then(res => res.json())
      .then(data => setExpedientesDepto(data));

    // carga inicial: todos los estatus
    fetch("http://localhost:3000/expedientes/estatus")
      .then(res => res.json())
      .then(data => setExpedientesEstatus(data));
  }, []);

  const dataDepto = {
    labels: expedientesDepto.map(d => d.departamento),
    datasets: [
      {
        label: "Expedientes por departamento",
        data: expedientesDepto.map(d => d.total),
        backgroundColor: "#3498DB",
      },
    ],
  };

  const optionsDepto = {
    onClick: async (evt, elements) => {
      if (!elements.length) return;
      const index = elements[0].index;
      const depto = dataDepto.labels[index];

      // fetch dinámico de estatus por departamento
      const res = await fetch(`http://localhost:3000/expedientes/estatus_depto/${depto}`);
      const data = await res.json();
      setExpedientesEstatus(data);
    },
  };

  const dataEstatus = {
    labels: expedientesEstatus.map(e => e.estatus),
    datasets: [
      {
        label: "Expedientes por estatus",
        data: expedientesEstatus.map(e => e.total),
        backgroundColor: ["#27AE60", "#C0392B", "#2980B9", "#F39C12", "#8E44AD"],
      },
    ],
  };

  const optionsEstatus = {
    plugins: {
      tooltip: {
        callbacks: {
          label: function (context) {
            const total = context.dataset.data.reduce((a, b) => a + b, 0);
            const value = context.raw;
            const porcentaje = ((value / total) * 100).toFixed(1);
            return `${context.label}: ${value} (${porcentaje}%)`;
          },
        },
      },
      legend: {
        position: "right",
        labels: {
          generateLabels: function (chart) {
            const data = chart.data;
            const total = data.datasets[0].data.reduce((a, b) => a + b, 0);
            return data.labels.map((label, i) => {
              const value = data.datasets[0].data[i];
              const porcentaje = ((value / total) * 100).toFixed(1);
              return {
                text: `${label} (${porcentaje}%)`,
                fillStyle: data.datasets[0].backgroundColor[i],
                index: i,
              };
            });
          },
        },
      },
    },
  };

  return (
    <div className="contenedor-principal">
      <h2 style={{ textAlign: "center", marginBottom: "20px" }}>
        Dashboard Interactivo
      </h2>

      <div style={{ display: "flex", gap: "40px", flexWrap: "wrap" }}>
        <div style={{ width: "80%", margin: "auto" }}>
          <h3>Expedientes por Departamento</h3>
          <Bar data={dataDepto} options={optionsDepto} />
        </div>

        <div style={{ width: "80%", margin: "auto" }}>
          <h3>Expedientes por Estatus</h3>
          <Pie data={dataEstatus} options={optionsEstatus} />
        </div>
      </div>
    </div>
  );
}

export default Dashboard;
