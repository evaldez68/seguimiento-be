// src/components/MaestroEnvioForm.jsx
import React, { useState, useEffect } from "react";

function MaestroEnvioForm({ onGrabar, maestroInicial }) {
  const [formData, setFormData] = useState({
    id_coordinacion: "",
    correlativo: "",
    tipo_entrega: "Presencial",
    no_guia: "",
    empresa_courier: "",
  });

  const [coordinaciones, setCoordinaciones] = useState([]);

  useEffect(() => {
    // Cargar coordinaciones
    fetch("http://localhost:3000/coordinaciones")
      .then((res) => res.json())
      .then((data) => setCoordinaciones(data));

    // Si viene maestroInicial (buscar nota), usarlo
    if (maestroInicial && maestroInicial.correlativo) {
      setFormData((prev) => ({ ...prev, ...maestroInicial }));
    } else {
      // Si es nueva nota, pedir correlativo al backend
      fetch("http://localhost:3000/maestro_envios/ultimo-correlativo")
        .then((res) => res.json())
        .then((info) => {
          setFormData((prev) => ({ ...prev, correlativo: info.siguiente }));
        });
    }
  }, [maestroInicial]);

  const handleChange = (e) => {
    setFormData({ ...formData, [e.target.name]: e.target.value });
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    const fecha = new Date();
    const payload = {
      ...formData,
      fecha: fecha.toISOString().split("T")[0],
      hora: fecha.toTimeString().split(" ")[0],
    };

    const res = await fetch("http://localhost:3000/maestro_envios", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload),
    });
    const data = await res.json();
    onGrabar(data);
  };

  return (
    <form onSubmit={handleSubmit} className="formulario">
      <h3>Ingreso Maestro de Envío</h3>

      <label>Coordinación:</label>
      <select
        name="id_coordinacion"
        value={formData.id_coordinacion}
        onChange={handleChange}
        required
      >
        <option value="">Seleccione coordinación</option>
        {coordinaciones.map((c) => (
          <option key={c.id} value={c.id}>
            {c.departamento}
          </option>
        ))}
      </select>

      <label>Correlativo:</label>
      <input type="text" name="correlativo" value={formData.correlativo} readOnly />

      <label>Tipo de Entrega:</label>
      <select name="tipo_entrega" value={formData.tipo_entrega} onChange={handleChange}>
        <option value="Presencial">Presencial</option>
        <option value="Courrier">Courier</option>
      </select>

      {formData.tipo_entrega === "Courier" && (
        <>
          <label>No. Guía:</label>
          <input type="text" name="no_guia" value={formData.no_guia} onChange={handleChange} />
          <label>Empresa Courier:</label>
          <input type="text" name="empresa_courier" value={formData.empresa_courier} onChange={handleChange} />
        </>
      )}

      <button type="submit" className="guardar">Grabar</button>
    </form>
  );
}

export default MaestroEnvioForm;

//src/components/ConsultaDocumentos.jsx

import React, { useState } from "react";
import {
  Box,
  TextField,
  Button,
  Select,
  MenuItem,
  InputLabel,
  FormControl,
  Table,
  TableHead,
  TableRow,
  TableCell,
  TableBody,
} from "@mui/material";
import dayjs from "dayjs";

const ConsultaDocumentos = () => {
  const [tipoConsulta, setTipoConsulta] = useState("documento"); // default: No. Documento
  const [numeroDoc, setNumeroDoc] = useState("");
  const [correlativo, setCorrelativo] = useState("");
  const [documentos, setDocumentos] = useState([]);
  const [listaDocs, setListaDocs] = useState([]);

  const buscarPorDocumento = async (num) => {
    const res = await fetch(`http://localhost:3000/seguimiento_consulta/documento/${num}`);
    const data = await res.json();
    setDocumentos(data);
  };

  const buscarPorCorrelativo = async () => {
    const res = await fetch(`http://localhost:3000/seguimiento_consulta/correlativo/${correlativo}`);
    const data = await res.json();
    setDocumentos(data);

    // 🔹 Filtrar duplicados de numero_documento
    const unicos = [...new Set(data.map(d => d.numero_documento))];
    setListaDocs(unicos);
  };

  // 🔹 Si el tipo es Documento, dispara la búsqueda automáticamente
  const handleNumeroDocChange = (e) => {
    const value = e.target.value;
    setNumeroDoc(value);
    if (tipoConsulta === "documento" && value.trim() !== "") {
      buscarPorDocumento(value);
    }
  };

  return (
    <Box sx={{ p: 2 }}>
      {/* Selector de tipo de consulta */}
      <FormControl sx={{ minWidth: 200, mb: 2 }}>
        <InputLabel>Tipo de Consulta</InputLabel>
        <Select
          value={tipoConsulta}
          onChange={(e) => {
            setTipoConsulta(e.target.value);
            setDocumentos([]);
            setListaDocs([]);
            setNumeroDoc("");
            setCorrelativo("");
          }}
        >
          <MenuItem value="documento">No. Documento</MenuItem>
          <MenuItem value="correlativo">Correlativo</MenuItem>
        </Select>
      </FormControl>

      {/* Campos según tipo de consulta */}
      {tipoConsulta === "documento" && (
        <TextField
          label="Número Documento"
          value={numeroDoc}
          onChange={handleNumeroDocChange}
          sx={{ mb: 2, width: "300px" }}
        />
      )}

      {tipoConsulta === "correlativo" && (
        <Box sx={{ display: "flex", gap: 2, mb: 2 }}>
          <TextField
            label="Correlativo"
            value={correlativo}
            onChange={(e) => setCorrelativo(e.target.value)}
          />
          <Button variant="contained" onClick={buscarPorCorrelativo}>
            Buscar
          </Button>

          {listaDocs.length > 0 && (
            <FormControl sx={{ minWidth: 200 }}>
              <InputLabel>Documentos</InputLabel>
              <Select
                value={numeroDoc}
                onChange={(e) => {
                  setNumeroDoc(e.target.value);
                  buscarPorDocumento(e.target.value);
                }}
              >
                {listaDocs.map((doc, idx) => (
                  <MenuItem key={idx} value={doc}>
                    {doc}
                  </MenuItem>
                ))}
              </Select>
            </FormControl>
          )}
        </Box>
      )}

      {/* GRID de resultados */}
      <Table>
        <TableHead>
          <TableRow>
            <TableCell>Usuario</TableCell>
            <TableCell>Fecha</TableCell>
            <TableCell>Hora GT</TableCell>
            <TableCell>Proceso</TableCell>
            <TableCell>Tipo Doc</TableCell>
            <TableCell>Número Doc</TableCell>
            <TableCell>Tipo Entrega</TableCell>
            <TableCell>No. Guía</TableCell>
            <TableCell>Courier</TableCell>
            <TableCell>Semáforo</TableCell>
          </TableRow>
        </TableHead>
        <TableBody>
          {documentos.map((row, idx) => {
            const fechaMovimiento = dayjs(`${row.fecha} ${row.horagt}`);
            const dias = dayjs().diff(fechaMovimiento, "day");

            let color = "green";
            if (dias >= 3 && dias <= 4) color = "gold";
            if (dias >= 5) color = "red";

            return (
              <TableRow key={idx}>
                <TableCell>{row.usuario}</TableCell>
                <TableCell>{row.fecha}</TableCell>
                <TableCell>{row.horagt}</TableCell>
                <TableCell>{row.nombre_proceso}</TableCell>
                <TableCell>{row.tipo_documento}</TableCell>
                <TableCell>{row.numero_documento}</TableCell>
                <TableCell>{row.tipo_entrega}</TableCell>
                <TableCell>{row.no_guia}</TableCell>
                <TableCell>{row.empresa_courier}</TableCell>
                <TableCell>
                  <Box
                    sx={{
                      width: 20,
                      height: 20,
                      borderRadius: "50%",
                      backgroundColor: color,
                      margin: "auto",
                    }}
                  />
                </TableCell>
              </TableRow>
            );
          })}
        </TableBody>
      </Table>
    </Box>
  );
};

export default ConsultaDocumentos;
